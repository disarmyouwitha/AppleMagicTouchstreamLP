cmake_minimum_required(VERSION 3.20)
project(AMTJavelin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB_RECURSE JAVELIN_SOURCES CONFIGURE_DEPENDS Vendor/*.cc)
list(FILTER JAVELIN_SOURCES EXCLUDE REGEX "/processor/fake_processor\\.cc$")
list(FILTER JAVELIN_SOURCES EXCLUDE REGEX "/thread\\.cc$")
add_library(AMTJavelin SHARED Bridge.cc ${JAVELIN_SOURCES})
target_include_directories(AMTJavelin PRIVATE Vendor)
target_compile_definitions(AMTJavelin PRIVATE "JAVELIN_BOARD_CONFIG=<stddef.h>" AMT_HOST_EMITTER=1)
if(WIN32)
  if(MSVC AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Javelin requires Clang GNU bitfield layout; use ClangCL or the documented Zig build.")
  endif()
  if(MSVC)
    target_compile_options(AMTJavelin PRIVATE /clang:-mno-ms-bitfields)
  else()
    target_compile_options(AMTJavelin PRIVATE -mno-ms-bitfields)
    target_link_options(AMTJavelin PRIVATE -s)
  endif()
  set_target_properties(AMTJavelin PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON PREFIX "")
  target_compile_definitions(AMTJavelin PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX strdup=_strdup)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(AMTJavelin PRIVATE -Wno-unknown-attributes -Wno-unused-value)
endif()
